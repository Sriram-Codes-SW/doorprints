package app.doorprints.drive.photo

import app.doorprints.drive.DriveLayout
import app.doorprints.drive.NewFile
import app.doorprints.drive.delete.DeletionRules
import app.doorprints.drive.delete.ItemKind
import app.doorprints.drive.sync.SyncWorld
import kotlinx.coroutines.test.runTest
import kotlinx.io.Buffer
import org.junit.Assert.assertEquals
import org.junit.Assert.assertNull
import org.junit.Test

/**
 * S4b-BL-128 x S4b-BL-119: the files the photo service writes are classed as ours by `DeletionRules.classifyFile` (kind
 * `photo`, in the folder with role `photos`), so a deletion run removes them; a file with our kind in another folder, or
 * without a kind, is foreign. Needs `feat/drive-deletion` (`drive/delete/DeletionRules.kt`), so it is kept out of the
 * photos branch's sources until the two are merged (see the notes).
 */
class DeletionClassifiesPhotosTest {
    @Test
    fun photoFilesAreOurs() = runTest {
        val w = SyncWorld()
        val a = w.add("A")
        val bytes = ByteArray(3000) { it.toByte() }
        a.photos().upload("p1", bytes.size.toLong()) { Buffer().apply { write(bytes) } }
        val folderRole = w.server.fileOrNull(w.photosFolderId())!!.appProperties.getValue(DriveLayout.ROLE)
        assertEquals("photos", folderRole)
        val file = w.photoFiles().single()
        assertEquals(ItemKind.PHOTO, DeletionRules.classifyFile(file.appProperties, folderRole))
        // A partial photo (a stopped upload) carries the same kind and is ours too: the run removes leftovers.
        val partial = w.server.putByHand(
            NewFile("partial-p-x.dpx", "application/octet-stream", listOf(w.photosFolderId()), file.appProperties + (DriveLayout.STATE to DriveLayout.STATE_PARTIAL)),
        )
        assertEquals(ItemKind.PHOTO, DeletionRules.classifyFile(partial.appProperties, folderRole))
        // Foreign: the same properties outside Photos, a photo kind in Backups, and a file with no kind.
        assertNull(DeletionRules.classifyFile(file.appProperties, "backups"))
        assertNull(DeletionRules.classifyFile(file.appProperties, "root"))
        assertNull(DeletionRules.classifyFile(mapOf(PROP_PHOTO_ID to "p1"), folderRole))
        assertEquals(2, DeletionRules.phaseOf(ItemKind.PHOTO))
    }
}
